290 template <class T> inline Sign ratio_compare(
291 const T& a_num, const T& a_denom, const T& b_num, const T& b_denom
292 ) {
293 if(a_denom == b_denom) {
294 return Sign(geo_cmp(a_num,b_num)*geo_sgn(a_denom));
295 }
296 return Sign(
297 geo_cmp(a_num*b_denom, b_num*a_denom) *
298 geo_sgn(a_denom) * geo_sgn(b_denom)
299 );
300 }

429// GCC does not have any pragma to deactivate FMA generation,
430// so instead we check that they are deactivated (by the command-line
431// option -ffp-contract=off) and fire an assertion fail if it was not
432// the case. If targeting (very old) x87, deactivates the old extended
433// precision mode .
434 struct GeoAssertStrictIEEE754 {
435 GeoAssertStrictIEEE754() {
436#ifdef GEO_387 // disable (very old) 387 extended precision
437 geo_disable_387_extended_precision();
438#endif
439#ifdef GEOGRAM_PSM
440 if(fp_contraction_enabled()) {
441 std::cerr << "Needs to be compiled with -ffp-contract-off "
442 << "(or no x87 extended precision)"
443 << std::endl;
444 abort();
445 }
446#else
447 geo_assert(!fp_contraction_enabled());
448#endif
449 }
450 static bool fp_contraction_enabled() {
451 return (a2plusb(0x1.0000002p0, -0x1.0000004p0) != 0.0);
452 }
453 __attribute__((noipa)) static double a2plusb(double a, double b) {
454 return a * a + b;
455 }
456 };
457}
