255 template <class T> class GenericLifeCycle : public LifeCycle {
256 public:
257
261 GenericLifeCycle() :
262 LifeCycle(
263 sizeof(T),
264 (std::is_standard_layout<T>::value && std::is_trivial<T>::value)
265 ) {
266 }
267
271 void construct(Memory::pointer address) override {
272 new(address)T;
273 }
274
278 void copy_construct(
279 Memory::pointer lhs, Memory::const_pointer rhs
280 ) override {
281 new(lhs)T(*(const T*)rhs);
282 }
283
287 void destroy(Memory::pointer address) override {
288 geo_argused(address); // Silences a MSVC warning.
289 ((T*)address)->~T();
290 }
291
295 void assign(Memory::pointer lhs, Memory::const_pointer rhs) override {
296 *(T*)lhs=*(const T*)rhs;
297 }
298
302 void reset(Memory::pointer address) override {
303 *(T*)address = T();
304 }
305
308 void swap(Memory::pointer a, Memory::pointer b) override {
309 std::swap(*(T*)a, *(T*)b);
310 }
311
315 void construct_array(Memory::pointer address, index_t nb) override {
316 T* array = (T*)address;
317 for(index_t i=0; i<nb; ++i) {
318 new(&array[i])T;
319 }
320 }
321
325 void copy_construct_array(
326 Memory::pointer lhs, Memory::const_pointer rhs, index_t nb
327 ) override {
328 T* lhs_array = (T*)lhs;
329 const T* rhs_array = (const T*)rhs;
330 for(index_t i=0; i<nb; ++i) {
331 new(&lhs_array[i])T(rhs_array[i]);
332 }
333 }
334
338 void destroy_array(Memory::pointer address, index_t nb) override {
339 T* array = (T*)address;
340 geo_argused(array); // Silences a MSVC warning.
341 for(index_t i=0; i<nb; ++i) {
342 array[i].~T();
343 }
344 }
345
349 void assign_array(
350 Memory::pointer lhs, Memory::const_pointer rhs, index_t nb
351 ) override {
352 T* lhs_array = (T*)lhs;
353 const T* rhs_array = (const T*)rhs;
354 for(index_t i=0; i<nb; ++i) {
355 lhs_array[i] = rhs_array[i];
356 }
357 }
358
362 void reset_array(Memory::pointer address, index_t nb) override {
363 T* array = (T*)address;
364 geo_argused(array); // Silences a MSVC warning.
365 for(index_t i=0; i<nb; ++i) {
366 array[i] = T();
367 }
368 }
369
373 Memory::pointer new_object() override {
374 return Memory::pointer(new T);
375 }
376
380 Memory::pointer new_object(Memory::const_pointer rhs) override {
381 return Memory::pointer(new T(*(const T*)rhs));
382 }
383
387 void delete_object(Memory::pointer address) override {
388 delete((T*)address);
389 }
390
394 Memory::pointer new_array(index_t nb) override {
395 return Memory::pointer(new T[nb]);
396 }
397
401 void delete_array(Memory::pointer address) override {
402 delete[]((T*)address);
403 }
404
405 };
