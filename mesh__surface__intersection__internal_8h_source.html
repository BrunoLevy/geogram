455 void flip() {
456 std::swap(f1,f2);
457 A_rgn_f1 = swap_T1_T2(A_rgn_f1);
458 A_rgn_f2 = swap_T1_T2(A_rgn_f2);

460 B_rgn_f1 = swap_T1_T2(B_rgn_f1);
461 B_rgn_f2 = swap_T1_T2(B_rgn_f2);
462 std::swap(B_rgn_f1, B_rgn_f2);
463 }

648 Halfedges(
649 CoplanarFacets& coplanar_facets

652
657 void initialize() {
658 // Use resize rather than assign so that we do not traverse
659 // all the halfedges of the mesh_
660 v_first_halfedge_.resize(mesh_.vertices.nb(), NO_INDEX);
661 h_next_around_v_.resize(mesh_.facet_corners.nb(), NO_INDEX);
662 // We only need to reset the halfedges of this set of coplanar
663 // facets.
664 for(index_t h: halfedges_) {
665 v_first_halfedge_[vertex(h,0)] = NO_INDEX;
666 h_next_around_v_[h] = NO_INDEX;
667 }
668 halfedges_.resize(0);
669 }
670
677 index_t vertex(index_t h, index_t dlv) const {
678 index_t f = h/3;
679 index_t lv = (h+dlv)%3;
680 return mesh_.facets.vertex(f,lv);
681 }
682
688 index_t facet(index_t h) const {
689 return h/3;
690 }
691
698 index_t alpha2(index_t h) const {
699 index_t t1 = facet(h);
700 index_t t2 = mesh_.facet_corners.adjacent_facet(h);
701 if(t2 == NO_INDEX) {
702 return NO_INDEX;
703 }
704 for(index_t h2: mesh_.facets.corners(t2)) {
705 if(mesh_.facet_corners.adjacent_facet(h2) == t1) {
706 return h2;
707 }
708 }
709 geo_assert_not_reached;
710 }
711
716 void add(index_t h) {
717 halfedges_.push_back(h);
718 index_t v1 = vertex(h,0);
719 h_next_around_v_[h] = v_first_halfedge_[v1];
720 v_first_halfedge_[v1] = h;
721 }
722
727 vector<index_t>::const_iterator begin() const {
728 return halfedges_.begin();
729 }
730
735 vector<index_t>::const_iterator end() const {
736 return halfedges_.end();
737 }
738
744 index_t vertex_first_halfedge(index_t v) const {
745 return v_first_halfedge_[v];
746 }
747
754 index_t next_around_vertex(index_t h) const {
755 return h_next_around_v_[h];
756 }
757
763 index_t nb_halfedges_around_vertex(index_t v) const {
764 index_t result = 0;
765 for(
766 index_t h = vertex_first_halfedge(v);
767 h != NO_INDEX;
768 h = next_around_vertex(h)
769 ) {
770 ++result;
771 }
772 return result;
773 }
774
784 index_t next_along_polyline(index_t h) const {
785 index_t v2 = vertex(h,1);
786 if(nb_halfedges_around_vertex(v2) != 1) {
787 return NO_INDEX;
788 }
789 return vertex_first_halfedge(v2);
790 }

792 private:
793 Mesh& mesh_;
794 vector<index_t> halfedges_;
795 vector<index_t> v_first_halfedge_;
796 vector<index_t> h_next_around_v_;
797 } halfedges_;
