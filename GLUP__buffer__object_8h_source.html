174 template <class T> void create_or_update(
175 index_t nb, const T* data, GLuint unit=0
176 ) {
177 if(unit != 0) {
178 active_unit(unit);
179 }
180 Type type = default_type();
181 switch(type) {
182 case UNINITIALIZED:
183 geo_assert_not_reached;
184 break;
185 case TEXTURE_2D:
186 create_or_update_TEXTURE_2D(nb, data);
187 break;
188 case TBO:
189 create_or_update_TBO(nb,data);
190 break;
191 case SSBO:
192 create_or_update_SSBO(nb,data);
193 break;
194 }
195 if(unit != 0) {
196 active_unit(0);
197 }
198 }

268 template <class T> void create_or_update_TEXTURE_2D(
269 index_t nb, const T* data
270 ) {
271 geo_debug_assert(type_ == TEXTURE_2D || type_ == UNINITIALIZED);
272 type_ = TEXTURE_2D;
273 if(texture_ == 0) {
274 glGenTextures(1, &texture_);
275 }
276 glBindTexture(GL_TEXTURE_2D, texture_);
277 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_NEAREST);
278 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_NEAREST);
279 GLsizei width = GLsizei(nb);
280 GLsizei height = 1;
281 if(nb > 1024) {
282 width = GLsizei(::sqrt(nb));
283 height = GLsizei(nb) / width;
284 while(width*height < GLsizei(nb)) {
285 ++height;
286 }
287 }
288 glPixelStorei(GL_PACK_ALIGNMENT, 1);
289 if(width_ != width || height_ != height) {
290 glTexImage2D(
291 GL_TEXTURE_2D,
292 0, // level
293 GLint(internal_format(T())),
294 width, height,
295 0, // border
296 format(T()), type(T()),
297 nullptr
298 );
299 width_ = width;
300 height_ = height;
301 }
302 if(GLsizei(nb) == width*height) {
303 glTexSubImage2D(
304 GL_TEXTURE_2D,
305 0, // level
306 0, 0, // xoffset, yoffset
307 width, height,
308 format(T()), type(T()),
309 data
310 );
311 } else {
312 // If texture's last row is slightly larger than input
313 // data, avoid reading past the end of data by transfering
314 // the (partial) last row:
315 glTexSubImage2D(
316 GL_TEXTURE_2D,
317 0, // level
318 0, 0, // xoffset, yoffset
319 width, height-1, // width, height
320 format(T()), type(T()),
321 data
322 );
323 glTexSubImage2D(
324 GL_TEXTURE_2D,
325 0, // level
326 0, height-1, // xoffset, yoffset
327 (GLsizei(nb) % width), 1, // width, height
328 format(T()), type(T()),
329 data + (height-1)*width
330 );
331 }
332 }

334 template <class T> GEO_TBO_FUNC void create_or_update_TBO(
335 index_t nb, const T* data
336 ){
337#ifdef GEO_TBO_SUPPORTED
338 geo_debug_assert(type_ == TBO || type_ == UNINITIALIZED);
339 type_ = TBO;
340 geo_assert(GLint(nb) < max_TBO_size_);
341 update_or_check_buffer_object(
342 BO_, GL_ARRAY_BUFFER,
343 nb * sizeof(T), data,
344 true
345 );
346 if(texture_ == 0) {
347 glGenTextures(1, &texture_);
348 glBindTexture(GL_TEXTURE_BUFFER, texture_);
349 glTexBuffer(GL_TEXTURE_BUFFER, internal_format(T()), BO_);
350 }
351#else
352 geo_argused(nb);
353 geo_argused(data);
354 geo_assert_not_reached;
355#endif
356 }
