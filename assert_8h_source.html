137// Three levels of assert:
138// use geo_assert() and geo_range_assert() non-expensive asserts
139// use geo_debug_assert() and geo_debug_range_assert() expensive asserts
140// use geo_parano_assert() and geo_parano_range_assert() very exensive asserts
141
142
143// used to implement geo_xxx_assert() macros: puts argument in a block and
144// has a dummy while to avoid compiler warning about spurious semi-colon.
145#define GEO_ASSERTION(X) do { X } while(false)
146#define GEO_NO_ASSERTION() do { } while(false)
147
155#define geo_assert(x) GEO_ASSERTION({ \
156 if(!(x)) { \
157 GEO::geo_assertion_failed(#x, __FILE__, __LINE__); \
158 } \
159 })
160
171#define geo_range_assert(x, min_val, max_val) GEO_ASSERTION({ \
172 if(((x) < (min_val)) || ((x) > (max_val))) { \
173 GEO::geo_range_assertion_failed(x, min_val, max_val, \
174 __FILE__, __LINE__ \
175 ); \
176 } \
177 })
178
183#define geo_assert_not_reached GEO_ASSERTION({ \
184 GEO::geo_should_not_have_reached(__FILE__, __LINE__); \
185 })
