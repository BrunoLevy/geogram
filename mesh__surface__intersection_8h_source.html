458 std::tuple<vec3, vec3, vec3> get_initial_facet_vertices(index_t f) const {
459 // All facets are duplicated before radial sort. If f is one of
460 // the duplicated facets then its orientation is flipped as
461 // compared to initial facet.
462 index_t orig_f = original_facet_id_[f];
463 vec3 p1 = mesh_copy_.facets.point(orig_f,0);
464 vec3 p2 = mesh_copy_.facets.point(orig_f,1);
465 vec3 p3 = mesh_copy_.facets.point(orig_f,2);
466 if(initial_facet_is_flipped(f)) {
467 std::swap(p1,p3);
468 }
469 return std::make_tuple(p1,p2,p3);
470 }

985 bool radial_sort(index_t bndl, RadialSort& RS) {
986 geo_debug_assert(bndl < nb());
987 if(nb_halfedges(bndl) <= 2) {
988 return true;
989 }
990 auto b = H_.begin() + std::ptrdiff_t(bndl_start_[bndl]);
991 auto e = H_.begin() + std::ptrdiff_t(bndl_start_[bndl+1]);
992 RS.init(*b);
993 std::sort(
994 b, e, [&RS](index_t h1, index_t h2) {
995 return RS(h1,h2);
996 }
997 );
998 bool OK = !RS.degenerate();
999 bndl_is_sorted_[bndl] = OK;
1000 return OK;
1001 }
